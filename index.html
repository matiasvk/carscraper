Nettiauto Selaaja • Toteutettu Node.js-taustapalvelulla ja JSON-LD-jäsentimellä
server.js-määritys 
const express = require('express');
const axios = require('axios');
const cheerio = require('cheerio');
const cors = require('cors'); // npm install cors

const app = express();
app.use(cors()); // Salli yhteydet selaimesta

app.get('/api/cars/:make/:model', async (req, res) => {
    const { make, model } = req.params;
    const url = `https://www.nettiauto.com/${make}/${model}`;

    try {
        const response = await axios.get(url, {
            headers: { 'User-Agent': 'Mozilla/5.0' }
        });
        const $ = cheerio.load(response.data);
        let carsData = [];

        $('script[type="application/ld+json"]').each((_, el) => {
            try {
                const json = JSON.parse($(el).html());
                if (json.mainEntity && json.mainEntity.itemListElement) {
                    carsData = json.mainEntity.itemListElement.map(e => ({
                        name: e.item.name,
                        price: e.item.offers ? e.item.offers.price : null,
                        currency: e.item.offers ? e.item.offers.priceCurrency : 'EUR',
                        mileageKm: e.item.mileageFromOdometer?.value || 0,
                        fuelType: e.item.fuelType,
                        transmission: e.item.vehicleTransmission,
                        bodyType: e.item.bodyType,
                        color: e.item.color,
                        image: e.item.image,
                        url: e.item.url
                    }));
                }
            } catch(err) {}
        });

        res.json({ success: true, count: carsData.length, data: carsData });
    } catch (e) {
        res.status(500).json({ success: false, error: 'Tietojen haku epäonnistui' });
    }
});

app.listen(3000, () => console.log('Palvelin käynnissä portissa 3000'));
